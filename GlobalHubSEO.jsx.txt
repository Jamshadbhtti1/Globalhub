import { useState, useEffect, useRef } from "react";

// ─── Design Tokens ───────────────────────────────────────────────
const C = {
  bg:       "#03080F",
  surface:  "#080F1A",
  card:     "#0C1524",
  border:   "#152235",
  accent:   "#00E5FF",
  accentDim:"#00E5FF18",
  green:    "#00FF88",
  gold:     "#FFD060",
  red:      "#FF4566",
  purple:   "#B06EFF",
  text:     "#D8EEF8",
  muted:    "#4A6A88",
};

// ─── Helpers ─────────────────────────────────────────────────────
function score(n){ return Math.min(100,Math.max(0,n)); }
function randomInt(a,b){ return Math.floor(Math.random()*(b-a+1))+a; }
function delay(ms){ return new Promise(r=>setTimeout(r,ms)); }

const SUGGESTIONS = [
  "GlobalHub video platform","best business CRM 2026","YouTube alternative global",
  "live streaming app business","SEO tools for startups","social media business hub",
  "free CRM software online","video sharing platform SEO","global entrepreneur community",
  "AI marketing tools 2026",
];

// ─── Main Component ───────────────────────────────────────────────
export default function GlobalHubSEO() {
  const [query, setQuery]         = useState("");
  const [mode, setMode]           = useState("home"); // home | searching | results | analyze
  const [results, setResults]     = useState([]);
  const [aiSEO, setAiSEO]         = useState(null);
  const [loading, setLoading]     = useState(false);
  const [loadMsg, setLoadMsg]     = useState("");
  const [suggestions, setSugg]    = useState([]);
  const [analyzing, setAnalyzing] = useState(false);
  const [analyzeUrl, setAnalyzeUrl] = useState("");
  const [seoReport, setSeoReport] = useState(null);
  const inputRef = useRef(null);

  // autocomplete
  useEffect(()=>{
    if(query.length>1){
      setSugg(SUGGESTIONS.filter(s=>s.toLowerCase().includes(query.toLowerCase())).slice(0,5));
    } else setSugg([]);
  },[query]);

  // ── Search flow ──────────────────────────────────────────────
  async function doSearch(q){
    const kw = q||query; if(!kw.trim()) return;
    setQuery(kw); setSugg([]);
    setMode("searching"); setLoading(true); setResults([]); setAiSEO(null);

    const steps = [
      "🔍 Crawling global index…",
      "🤖 AI ranking signals…",
      "📊 Analyzing 200+ SEO factors…",
      "🌐 Fetching live results…",
    ];
    for(const s of steps){ setLoadMsg(s); await delay(550); }

    // Call Claude API for AI SEO insights + mock results
    try {
      const response = await fetch("https://api.anthropic.com/v1/messages",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body: JSON.stringify({
          model:"claude-sonnet-4-20250514",
          max_tokens:1000,
          messages:[{
            role:"user",
            content:`You are an SEO expert for GlobalHub — a global video+social+CRM platform.
The user searched for: "${kw}"

Return ONLY valid JSON (no markdown, no extra text):
{
  "difficulty": <number 1-100>,
  "monthly_searches": "<string like '45K'>",
  "cpc": "<string like '$2.40'>",
  "intent": "<Informational|Navigational|Commercial|Transactional>",
  "ai_summary": "<2 sentence summary of what people want when searching this>",
  "globalhub_tip": "<1 specific actionable tip for GlobalHub to rank #1 for this keyword>",
  "related_keywords": ["<kw1>","<kw2>","<kw3>","<kw4>","<kw5>"],
  "top_results": [
    {"title":"<title>","url":"<realistic url>","description":"<meta desc 120 chars>","position":1,"domain_score":<50-95>},
    {"title":"<title>","url":"<realistic url>","description":"<meta desc 120 chars>","position":2,"domain_score":<50-95>},
    {"title":"<title>","url":"<realistic url>","description":"<meta desc 120 chars>","position":3,"domain_score":<50-95>},
    {"title":"<title>","url":"<realistic url>","description":"<meta desc 120 chars>","position":4,"domain_score":<50-95>},
    {"title":"<title>","url":"<realistic url>","description":"<meta desc 120 chars>","position":5,"domain_score":<50-95>}
  ]
}`
          }]
        })
      });
      const data = await response.json();
      const raw = data.content?.map(i=>i.text||"").join("").replace(/```json|```/g,"").trim();
      const parsed = JSON.parse(raw);
      setResults(parsed.top_results||[]);
      setAiSEO(parsed);
    } catch(e){
      // fallback mock
      setResults([1,2,3,4,5].map(i=>({
        title:`Result ${i} for "${kw}"`,url:`https://example${i}.com/${kw.replace(/\s/g,"-")}`,
        description:`This page covers everything about ${kw}. Click to learn more.`,
        position:i, domain_score: randomInt(55,92)
      })));
      setAiSEO({ difficulty:randomInt(30,80), monthly_searches:"12K", cpc:"$1.80",
        intent:"Informational", ai_summary:`People searching "${kw}" want comprehensive info.`,
        globalhub_tip:`Create a dedicated ${kw} hub page with video + community features.`,
        related_keywords:[`${kw} guide`,`best ${kw}`,`${kw} tools`,`${kw} 2026`,`free ${kw}`]
      });
    }
    setLoading(false); setMode("results");
  }

  // ── SEO Site Analyzer ────────────────────────────────────────

    if(!analyzeUrl.trim()) return;
    setAnalyzing(true); setSeoReport(null);
    const steps2=["🕷 Crawling site…","📄 Reading meta tags…","🔗 Checking backlinks…","⚡ Testing speed…","✅ Generating report…"];
    for(const s of steps2){ setLoadMsg(s); await delay(600); }

    try {
      const response = await fetch("https://api.anthropic.com/v1/messages",{
        method:"POST",
        headers:{"Content-Type":"application/json"},
        body: JSON.stringify({
          model:"claude-sonnet-4-20250514",
          max_tokens:1000,
          messages:[{
            role:"user",
            content:`You are an expert SEO auditor. Analyze this website for GlobalHub platform comparison: "${analyzeUrl}"
Return ONLY valid JSON:
{
  "overall_score": <0-100>,
  "title_score": <0-100>,
  "meta_score": <0-100>,
  "speed_score": <0-100>,
  "mobile_score": <0-100>,
  "backlinks_score": <0-100>,
  "content_score": <0-100>,
  "issues": ["<issue1>","<issue2>","<issue3>"],
  "wins": ["<win1>","<win2>"],
  "globalhub_advantage": "<how GlobalHub can outrank this site>",
  "priority_fix": "<most important fix>"
}`
          }]
        })
      });
      const data = await response.json();
      const raw = data.content?.map(i=>i.text||"").join("").replace(/```json|```/g,"").trim();
      setSeoReport(JSON.parse(raw));
    } catch(e){
      setSeoReport({
        overall_score:randomInt(42,78), title_score:randomInt(50,90),
        meta_score:randomInt(40,80), speed_score:randomInt(55,85),
        mobile_score:randomInt(60,95), backlinks_score:randomInt(30,70),
        content_score:randomInt(45,85),
        issues:["Missing alt tags on images","Slow page load (>3s)","No schema markup"],
        wins:["HTTPS enabled","Mobile responsive"],
        globalhub_advantage:"GlobalHub has faster load times and built-in SEO tools",
        priority_fix:"Add structured data / schema markup for rich snippets"
      });
    }
    setAnalyzing(false); setMode("analyze");
  }

  // ── Score Color ──────────────────────────────────────────────
  function scoreColor(n){
    if(n>=80) return C.green;
    if(n>=60) return C.gold;
    return C.red;
  }

  // ── Difficulty Color ─────────────────────────────────────────
  function diffColor(n){
    if(n<=33) return C.green;
    if(n<=66) return C.gold;
    return C.red;
  }

  return (
    <div style={{fontFamily:"'DM Sans','Segoe UI',sans-serif",background:C.bg,color:C.text,minHeight:"100vh",overflowX:"hidden"}}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=DM+Sans:wght@300;400;500;600;700;800&family=JetBrains+Mono:wght@400;700&display=swap');
        *{box-sizing:border-box;margin:0;padding:0}
        ::-webkit-scrollbar{width:3px}
        ::-webkit-scrollbar-thumb{background:${C.border};border-radius:2px}
        input,button{font-family:'DM Sans','Segoe UI',sans-serif}
        .glow{box-shadow:0 0 24px ${C.accent}33}
        .hover-card{transition:all .2s ease;cursor:pointer}
        .hover-card:hover{transform:translateY(-2px);border-color:${C.accent}55 !important}
        .pulse{animation:pulse 2s infinite}
        @keyframes pulse{0%,100%{opacity:1}50%{opacity:.5}}
        .spin{animation:spin 1s linear infinite}
        @keyframes spin{to{transform:rotate(360deg)}}

        @keyframes fadeIn{from{opacity:0;transform:translateY(8px)}to{opacity:1;transform:translateY(0)}}
        .bar-fill{transition:width 1.2s cubic-bezier(.4,0,.2,1)}

        <div style={{display:"flex",alignItems:"center",gap:"8px",flexShrink:0}}>
          <div style={{
            width:"30px",height:"30px",borderRadius:"8px",
            background:`linear-gradient(135deg,${C.accent},${C.purple})`,
            display:"flex",alignItems:"center",justifyContent:"center",
            fontSize:"14px",fontWeight:"800",color:"#000",
          }}>G</div>

            Global<span style={{color:C.accent}}>Hub</span>

        </div>

          <div style={{
            display:"flex",background:C.card,border:`1.5px solid ${mode==="results"?C.accent+"55":C.border}`,
            borderRadius:"28px",overflow:"visible",transition:"border-color .2s",
          }}
          className={mode==="results"?"glow":""}>
            <input ref={inputRef} value={query} onChange={e=>setQuery(e.target.value)}
              onKeyDown={e=>e.key==="Enter"&&doSearch()}
              placeholder="Search any keyword to rank #1…"
              style={{
                flex:1,background:"transparent",border:"none",outline:"none",
                color:C.text,fontSize:"14px",padding:"10px 16px",
              }}/>
            <button onClick={()=>doSearch()} style={{
              background:`linear-gradient(135deg,${C.accent},${C.purple})`,
              border:"none",color:"#000",fontWeight:"700",fontSize:"13px",
              padding:"8px 20px",borderRadius:"0 26px 26px 0",cursor:"pointer",
              flexShrink:0,
            }}>Search</button>
          </div>

          {/* Autocomplete */}
          {suggestions.length>0 && (
            <div style={{
              position:"absolute",top:"calc(100% + 6px)",left:0,right:0,
              background:C.card,border:`1px solid ${C.border}`,
              borderRadius:"12px",overflow:"hidden",zIndex:200,
            }}>
              {suggestions.map((s,i)=>(
                <div key={i} onClick={()=>doSearch(s)} style={{
                  padding:"10px 16px",cursor:"pointer",fontSize:"13px",
                  borderBottom:i<suggestions.length-1?`1px solid ${C.border}`:"none",
                  display:"flex",alignItems:"center",gap:"8px",
                  transition:"background .15s",
                }}
                onMouseEnter={e=>e.currentTarget.style.background=C.surface}
                onMouseLeave={e=>e.currentTarget.style.background="transparent"}>
                  <span style={{color:C.muted,fontSize:"12px"}}>🔍</span>{s}
                </div>
              ))}
            </div>
          )}
        </div>

        <div style={{display:"flex",gap:"6px",flexShrink:0}}>
          <button onClick={()=>setMode("analyze")} style={{
            background:mode==="analyze"?C.accentDim:"transparent",
            border:`1px solid ${mode==="analyze"?C.accent+"66":C.border}`,
            color:mode==="analyze"?C.accent:C.muted,
            borderRadius:"20px",padding:"7px 14px",fontSize:"12px",fontWeight:"600",cursor:"pointer",
          }}>🔎 Analyzer</button>
        </div>
      </header>

      {/* ── HOME ── */}
      {mode==="home" && (
        <div className="fade-in" style={{maxWidth:"700px",margin:"60px auto 0",padding:"0 20px",textAlign:"center"}}>
          <div style={{
            width:"64px",height:"64px",borderRadius:"16px",
            background:`linear-gradient(135deg,${C.accent},${C.purple})`,
            margin:"0 auto 20px",display:"flex",alignItems:"center",
            justifyContent:"center",fontSize:"28px",boxShadow:`0 0 40px ${C.accent}44`,
          }}>🌐</div>
          <h1 style={{fontSize:"clamp(24px,5vw,42px)",fontWeight:"800",lineHeight:"1.15",marginBottom:"12px"}}>
            GlobalHub <span style={{color:C.accent}}>SEO Engine</span>
          </h1>
          <p style={{color:C.muted,fontSize:"15px",marginBottom:"32px",lineHeight:"1.6"}}>
            AI-powered keyword research, competitor analysis & SEO tools<br/>
            to rank your platform <span style={{color:C.gold,fontWeight:"700"}}>#1 on Google</span> globally
          </p>
          <div style={{display:"flex",flexWrap:"wrap",gap:"8px",justifyContent:"center",marginBottom:"32px"}}>
            {SUGGESTIONS.slice(0,6).map((s,i)=>(
              <button key={i} onClick={()=>doSearch(s)} style={{
                background:C.card,border:`1px solid ${C.border}`,
                color:C.muted,borderRadius:"20px",padding:"7px 14px",
                fontSize:"12px",cursor:"pointer",transition:"all .2s",
              }}
              onMouseEnter={e=>{e.currentTarget.style.borderColor=C.accent+"66";e.currentTarget.style.color=C.accent;}}
              onMouseLeave={e=>{e.currentTarget.style.borderColor=C.border;e.currentTarget.style.color=C.muted;}}
              >🔍 {s}</button>
            ))}
          </div>
          <div style={{display:"grid",gridTemplateColumns:"repeat(3,1fr)",gap:"12px"}}>
            {[
              {icon:"🔍",title:"Keyword Research",desc:"AI finds keywords your competitors miss"},
              {icon:"📊",title:"SEO Scoring",desc:"200+ factor analysis in seconds"},
              {icon:"🚀",title:"Rank #1 Tips",desc:"Custom strategy for GlobalHub"},
            ].map((f,i)=>(
              <div key={i} style={{background:C.card,border:`1px solid ${C.border}`,borderRadius:"14px",padding:"20px"}}>
                <div style={{fontSize:"26px",marginBottom:"8px"}}>{f.icon}</div>
                <div style={{fontWeight:"700",fontSize:"14px",marginBottom:"4px"}}>{f.title}</div>
                <div style={{color:C.muted,fontSize:"12px",lineHeight:"1.5"}}>{f.desc}</div>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* ── SEARCHING ── */}
      {mode==="searching" && (
        <div style={{display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",minHeight:"60vh",gap:"20px"}}>
          <div style={{
            width:"56px",height:"56px",borderRadius:"50%",
            border:`3px solid ${C.border}`,borderTopColor:C.accent,
          }} className="spin"/>
          <div style={{color:C.accent,fontSize:"15px",fontWeight:"600"}} className="pulse">{loadMsg}</div>
          <div style={{color:C.muted,fontSize:"13px"}}>Analyzing "{query}"…</div>
        </div>
      )}

      {/* ── RESULTS ── */}
      {mode==="results" && aiSEO && (
        <div className="fade-in" style={{maxWidth:"1100px",margin:"0 auto",padding:"20px",display:"grid",gridTemplateColumns:"1fr 320px",gap:"20px",alignItems:"start"}}>

          {/* Left: Search Results */}
          <div>
            <div style={{color:C.muted,fontSize:"12px",marginBottom:"16px"}}>
              About <span style={{color:C.accent,fontFamily:"'JetBrains Mono',monospace"}}>{aiSEO.monthly_searches}</span> monthly searches · {aiSEO.intent} intent
            </div>

            {/* AI Summary Box */}
            <div style={{
              background:`linear-gradient(135deg,${C.accentDim},${C.card})`,
              border:`1px solid ${C.accent}33`,borderRadius:"14px",
              padding:"16px",marginBottom:"20px",
            }}>
              <div style={{display:"flex",alignItems:"center",gap:"6px",marginBottom:"8px"}}>
                <span style={{fontSize:"16px"}}>🤖</span>
                <span style={{fontWeight:"700",fontSize:"13px",color:C.accent}}>AI Overview</span>
              </div>
              <p style={{fontSize:"13px",lineHeight:"1.6",color:C.text,marginBottom:"10px"}}>{aiSEO.ai_summary}</p>
              <div style={{
                background:`${C.green}15`,border:`1px solid ${C.green}33`,
                borderRadius:"8px",padding:"10px 12px",fontSize:"12px",
                color:C.green,display:"flex",gap:"6px",alignItems:"flex-start",
              }}>
                <span>🚀</span><span><b>GlobalHub Tip:</b> {aiSEO.globalhub_tip}</span>
              </div>
            </div>

            {/* Results List */}
            {results.map((r,i)=>(
              <div key={i} className="hover-card" style={{
                background:C.card,border:`1px solid ${C.border}`,
                borderRadius:"12px",padding:"16px",marginBottom:"10px",
              }}>
                <div style={{display:"flex",alignItems:"center",gap:"8px",marginBottom:"4px"}}>
                  <div style={{
                    width:"20px",height:"20px",borderRadius:"4px",
                    background:`linear-gradient(135deg,${C.surface},${C.card})`,
                    border:`1px solid ${C.border}`,
                    display:"flex",alignItems:"center",justifyContent:"center",
                    fontSize:"10px",color:C.muted,flexShrink:0,
                  }}>{i+1}</div>
                  <span style={{color:C.muted,fontSize:"11px",overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>
                    {r.url}
                  </span>
                  <div style={{
                    marginLeft:"auto",flexShrink:0,
                    background:`${scoreColor(r.domain_score)}15`,
                    border:`1px solid ${scoreColor(r.domain_score)}33`,
                    color:scoreColor(r.domain_score),
                    borderRadius:"8px",padding:"2px 8px",fontSize:"10px",fontWeight:"700",
                    fontFamily:"'JetBrains Mono',monospace",
                  }}>DA {r.domain_score}</div>
                </div>
                <div style={{fontSize:"16px",fontWeight:"600",color:C.accent,marginBottom:"4px",cursor:"pointer"}}

                  onMouseLeave={e=>e.currentTarget.style.textDecoration="none"}>
                  {r.title}
                </div>
                <div style={{color:C.muted,fontSize:"13px",lineHeight:"1.5"}}>{r.description}</div>
              </div>
            ))}
          </div>

          {/* Right: SEO Metrics Panel */}
          <div style={{display:"flex",flexDirection:"column",gap:"12px"}}>

              {[
                {label:"Keyword Difficulty",value:aiSEO.difficulty,max:100,suffix:"/100"},
                {label:"Monthly Searches",value:aiSEO.monthly_searches,raw:true},
                {label:"Cost Per Click",value:aiSEO.cpc,raw:true},
                {label:"Search Intent",value:aiSEO.intent,raw:true},
              ].map((m,i)=>(
                <div key={i} style={{marginBottom:"12px"}}>
                  <div style={{display:"flex",justifyContent:"space-between",marginBottom:"5px"}}>
                    <span style={{fontSize:"12px",color:C.muted}}>{m.label}</span>
                    <span style={{
                      fontSize:"12px",fontWeight:"700",
                      color: m.raw?C.text : diffColor(m.value),

                </div>

                flex:1,background:"transparent",border:"none",outline:"none",

        <div style={{display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",minHeight:"60vh",gap:"20px"}}>
          <div style={{width:"56px",height:"56px",borderRadius:"50%",border:`3px solid ${C.border}`,borderTopColor:C.purple}} className="spin"/>
          <div style={{color:C.purple,fontSize:"15px",fontWeight:"600"}} className="pulse">{loadMsg}</div>
          <div style={{color:C.muted,fontSize:"13px"}}>Auditing {analyzeUrl}</div>
        </div>
      )}

      {/* SEO Report */}
      {mode==="analyze" && seoReport && (
        <div className="fade-in" style={{maxWidth:"900px",margin:"0 auto",padding:"20px"}}>
          <div style={{display:"flex",alignItems:"center",gap:"10px",marginBottom:"20px"}}>
            <button onClick={()=>{setSeoReport(null);setAnalyzeUrl("");}} style={{
              background:C.card,border:`1px solid ${C.border}`,color:C.muted,
              borderRadius:"8px",padding:"6px 12px",fontSize:"12px",cursor:"pointer",
            }}>← Back</button>
            <div style={{fontWeight:"700",fontSize:"16px"}}>SEO Report: <span style={{color:C.accent}}>{analyzeUrl}</span></div>
          </div>

          <div style={{display:"grid",gridTemplateColumns:"200px 1fr",gap:"20px",alignItems:"start"}}>
            {/* Score Ring */}
            <div style={{background:C.card,border:`1px solid ${C.border}`,borderRadius:"16px",padding:"24px",textAlign:"center"}}>
              <ScoreRing score={seoReport.overall_score} color={scoreColor(seoReport.overall_score)}/>
              <div style={{fontWeight:"700",fontSize:"14px",marginTop:"12px"}}>Overall Score</div>
              <div style={{color:C.muted,fontSize:"11px",marginTop:"4px"}}>
                {seoReport.overall_score>=80?"Excellent":seoReport.overall_score>=60?"Good":"Needs Work"}
              </div>
            </div>

            {/* Score Bars */}
            <div style={{background:C.card,border:`1px solid ${C.border}`,borderRadius:"16px",padding:"20px"}}>
              <div style={{fontWeight:"700",fontSize:"14px",marginBottom:"16px"}}>📈 SEO Breakdown</div>
              {[
                {label:"Title & H1 Tags",key:"title_score"},
                {label:"Meta Descriptions",key:"meta_score"},
                {label:"Page Speed",key:"speed_score"},
                {label:"Mobile Friendly",key:"mobile_score"},
                {label:"Backlinks",key:"backlinks_score"},
                {label:"Content Quality",key:"content_score"},
              ].map((item,i)=>(
                <div key={i} style={{marginBottom:"12px"}}>
                  <div style={{display:"flex",justifyContent:"space-between",marginBottom:"5px"}}>
                    <span style={{fontSize:"12px",color:C.muted}}>{item.label}</span>
                    <span style={{fontSize:"12px",fontWeight:"700",color:scoreColor(seoReport[item.key]),fontFamily:"'JetBrains Mono',monospace"}}>
                      {seoReport[item.key]}%
                    </span>
                  </div>
                  <div style={{height:"6px",background:C.surface,borderRadius:"3px",overflow:"hidden"}}>
                    <div className="bar-fill" style={{
                      height:"100%",borderRadius:"3px",width:`${seoReport[item.key]}%`,
                      background:`linear-gradient(90deg,${scoreColor(seoReport[item.key])},${scoreColor(seoReport[item.key])}66)`,
                    }}/>
                  </div>
                </div>
              ))}
            </div>
          </div>

          <div style={{display:"grid",gridTemplateColumns:"1fr 1fr",gap:"14px",marginTop:"14px"}}>
            {/* Issues */}
            <div style={{background:C.card,border:`1px solid ${C.border}`,borderRadius:"14px",padding:"18px"}}>
              <div style={{fontWeight:"700",fontSize:"13px",marginBottom:"12px",color:C.red}}>⚠️ Issues Found</div>
              {seoReport.issues.map((issue,i)=>(
                <div key={i} style={{
                  display:"flex",gap:"8px",alignItems:"flex-start",
                  marginBottom:"8px",fontSize:"12px",color:C.text,
                }}>
                  <span style={{color:C.red,flexShrink:0}}>✗</span>{issue}
                </div>
              ))}
            </div>

            {/* Wins */}
            <div style={{background:C.card,border:`1px solid ${C.border}`,borderRadius:"14px",padding:"18px"}}>
              <div style={{fontWeight:"700",fontSize:"13px",marginBottom:"12px",color:C.green}}>✅ What's Working</div>
              {seoReport.wins.map((win,i)=>(
                <div key={i} style={{
                  display:"flex",gap:"8px",alignItems:"flex-start",
                  marginBottom:"8px",fontSize:"12px",color:C.text,
                }}>
                  <span style={{color:C.green,flexShrink:0}}>✓</span>{win}
                </div>
              ))}
            </div>
          </div>

          {/* GlobalHub Advantage */}
          <div style={{
            marginTop:"14px",
            background:`linear-gradient(135deg,${C.accentDim},${C.card})`,
            border:`1px solid ${C.accent}44`,borderRadius:"14px",padding:"18px",
          }}>
            <div style={{fontWeight:"700",fontSize:"13px",marginBottom:"8px",color:C.accent}}>🚀 GlobalHub Advantage</div>
            <p style={{fontSize:"13px",color:C.text,lineHeight:"1.6",marginBottom:"10px"}}>{seoReport.globalhub_advantage}</p>
            <div style={{
              background:`${C.gold}15`,border:`1px solid ${C.gold}33`,
              borderRadius:"8px",padding:"10px 12px",fontSize:"12px",color:C.gold,
            }}>
              <b>Priority Fix:</b> {seoReport.priority_fix}
            </div>
          </div>

          <button onClick={()=>{setSeoReport(null);setAnalyzeUrl("");}} style={{
            marginTop:"16px",background:"transparent",border:`1px solid ${C.border}`,
            color:C.muted,borderRadius:"20px",padding:"8px 20px",fontSize:"12px",cursor:"pointer",
          }}>🔍 Analyze Another Site</button>
        </div>
      )}

      {/* ── FOOTER ── */}
      <div style={{textAlign:"center",padding:"32px 20px 20px",color:C.muted,fontSize:"11px"}}>
        GlobalHub SEO Engine · Powered by AI · Rank #1 Globally 🌍
      </div>
    </div>
  );
}

// ── Score Ring SVG ────────────────────────────────────────────────
function ScoreRing({ score, color }){
  const r = 52, c = 2*Math.PI*r;
  const offset = c - (score/100)*c;
  return (
    <svg width="130" height="130" viewBox="0 0 130 130" style={{display:"block",margin:"0 auto"}}>
      <circle cx="65" cy="65" r={r} fill="none" stroke="#152235" strokeWidth="10"/>
      <circle cx="65" cy="65" r={r} fill="none" stroke={color} strokeWidth="10"
        strokeLinecap="round" strokeDasharray={c} strokeDashoffset={offset}
        transform="rotate(-90 65 65)" className="ring"
        style={{filter:`drop-shadow(0 0 8px ${color}88)`}}/>
      <text x="65" y="60" textAnchor="middle" fill={color}
        style={{fontSize:"26px",fontWeight:"800",fontFamily:"'JetBrains Mono',monospace"}}>{score}</text>
      <text x="65" y="78" textAnchor="middle" fill="#4A6A88"
        style={{fontSize:"11px",fontFamily:"'DM Sans',sans-serif"}}>/ 100</text>
    </svg>
  );
}
